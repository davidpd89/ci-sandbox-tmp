def is_replyable(thread) -> bool:
    if thread is None:
        return False

    post = getattr(thread, "post", None)
    if post is None:
        return False

    record = getattr(post, "record", None)
    if record is None or not getattr(record, "text", "").strip():
        return False

    return bool(post.uri and post.cid)
