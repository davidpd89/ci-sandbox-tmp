from datetime import datetime

    canonical_url: Optional[str]
    parent_external_id: Optional[str]
    author_external_id: Optional[str]
    author_display_name: Optional[str]
    created_at_utc: Optional[datetime]
    text: str
    engagement: dict
    raw: dict
